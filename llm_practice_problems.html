from datasets import load_dataset
from transformers import AutoTokenizer
import matplotlib.pyplot as plt
import pandas as pd
import re, collections

# CELL 2 — Load dataset
dataset = load_dataset("knkarthick/dialogsum")
print(dataset)
print("\n--- Sample entry ---")
print(dataset["train"][0])

# CELL 3 — Explore the data
df = dataset["train"].to_pandas()
print(f"Training samples: {len(df)}")
print(f"Columns: {df.columns.tolist()}")
print(df.head(3))

# CELL 4 — Data cleaning function
def clean_text(text):
    text = text.strip()
    # Remove speaker labels like "#Person1#:"
    text = re.sub(r'#\w+#:\s*', '', text)
    # Collapse multiple spaces/newlines
    text = re.sub(r'\s+', ' ', text)
    # Remove very short dialogues (less than 20 chars)
    return text if len(text) > 20 else None

df['clean_dialogue'] = df['dialogue'].apply(clean_text)
df['clean_summary']  = df['summary'].apply(clean_text)

# Drop rows where cleaning returned None
df = df.dropna(subset=['clean_dialogue', 'clean_summary'])
print(f"After cleaning: {len(df)} samples")

# CELL 5 — Tokenize with Flan-T5 tokenizer
tokenizer = AutoTokenizer.from_pretrained("google/flan-t5-base")

def tokenize_example(row):
    prompt   = f"Summarize this dialogue:\n{row['clean_dialogue']}"
    response = row['clean_summary']
    input_tokens  = tokenizer(prompt,   truncation=True, max_length=512)
    output_tokens = tokenizer(response, truncation=True, max_length=128)
    return {
        'input_len' : len(input_tokens['input_ids']),
        'output_len': len(output_tokens['input_ids'])
    }

df[['input_len', 'output_len']] = df.apply(tokenize_example, axis=1, result_type='expand')

# CELL 6 — Visualize token lengths
fig, axes = plt.subplots(1, 2, figsize=(12, 4))
axes[0].hist(df['input_len'], bins=40, color='steelblue', edgecolor='white')
axes[0].set_title('Dialogue Token Length Distribution')
axes[0].set_xlabel('Tokens')
axes[1].hist(df['output_len'], bins=40, color='tomato', edgecolor='white')
axes[1].set_title('Summary Token Length Distribution')
axes[1].set_xlabel('Tokens')
plt.tight_layout()
plt.show()

print(f"\nMean input tokens : {df['input_len'].mean():.1f}")
print(f"Mean output tokens: {df['output_len'].mean():.1f}")
print(f"Max input tokens  : {df['input_len'].max()}")
Expected Output

  • After cleaning: ~14,350+ samples retained
  • Two histograms showing token distributions
  • Mean input ~130 tokens, mean output ~30 tokens

# ============================================================
# PRACTICE PROBLEM 2: Fine-tune GPT-2 on Indian Food Recipes
# Runtime: ~45 min on Colab T4 GPU
# ============================================================

# CELL 1 — Setup
!pip install transformers datasets accelerate -q

from transformers import (
    GPT2LMHeadModel, GPT2Tokenizer,
    TextDataset, DataCollatorForLanguageModeling,
    Trainer, TrainingArguments
)
from datasets import load_dataset
import torch, textwrap

print(f"GPU available: {torch.cuda.is_available()}")
print(f"GPU: {torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'None'}")

# CELL 2 — Load tokenizer and base model
model_name = "gpt2"
tokenizer = GPT2Tokenizer.from_pretrained(model_name)
tokenizer.pad_token = tokenizer.eos_token   # GPT2 has no pad token

# CELL 3 — Load & prepare recipe dataset
# Using a small recipe text dataset from HuggingFace Hub
dataset = load_dataset("Hieu-Pham/indian_food_recipes", split="train")
print(f"Recipes loaded: {len(dataset)}")
print(dataset[0])

# Format each recipe into a clean prompt string
def format_recipe(example):
    text = (
        f"Recipe: {example.get('name','')}\n"
        f"Ingredients: {example.get('ingredients','')}\n"
        f"Instructions: {example.get('instructions','')}\n"
        f"<|endoftext|>"
    )
    return {"text": text}

dataset = dataset.map(format_recipe)

# Save to text file (TextDataset expects a file)
with open("recipes.txt", "w") as f:
    for item in dataset:
        f.write(item["text"] + "\n")
print("Saved recipes.txt")

# CELL 4 — Create token dataset
train_dataset = TextDataset(
    tokenizer   = tokenizer,
    file_path   = "recipes.txt",
    block_size  = 256
)
data_collator = DataCollatorForLanguageModeling(
    tokenizer = tokenizer,
    mlm       = False   # GPT2 uses causal LM, not masked LM
)
print(f"Training blocks: {len(train_dataset)}")

# CELL 5 — BEFORE fine-tuning: see what base GPT-2 generates
base_model = GPT2LMHeadModel.from_pretrained(model_name)
def generate(model, prompt, max_new_tokens=100):
    inputs = tokenizer(prompt, return_tensors="pt")
    with torch.no_grad():
        output = model.generate(
            **inputs, max_new_tokens=max_new_tokens,
            do_sample=True, temperature=0.8, pad_token_id=tokenizer.eos_token_id
        )
    return textwrap.fill(tokenizer.decode(output[0], skip_special_tokens=True), 80)

prompt = "Recipe: Butter Chicken\nIngredients:"
print("=== BASE GPT-2 (before fine-tuning) ===")
print(generate(base_model, prompt))

# CELL 6 — Fine-tune!
training_args = TrainingArguments(
    output_dir              = "./gpt2-recipes",
    num_train_epochs        = 3,           # Try 5 for better results
    per_device_train_batch_size = 4,
    save_steps              = 500,
    save_total_limit        = 1,
    logging_steps           = 50,
    learning_rate           = 5e-5,        # Standard for fine-tuning
    warmup_steps            = 100,
    prediction_loss_only    = True,
    fp16                    = torch.cuda.is_available(),  # Mixed precision on GPU
)

ft_model = GPT2LMHeadModel.from_pretrained(model_name)
trainer = Trainer(
    model           = ft_model,
    args            = training_args,
    data_collator   = data_collator,
    train_dataset   = train_dataset,
)
trainer.train()
print("✅ Fine-tuning complete!")

# CELL 7 — AFTER fine-tuning: see what changed
print("\n=== FINE-TUNED GPT-2 (after training on recipes) ===")
print(generate(ft_model, prompt))

# Try different prompts!
for dish in ["Biryani", "Samosa", "Dal Makhani"]:
    print(f"\n--- {dish} ---")
    print(generate(ft_model, f"Recipe: {dish}\nIngredients:"))
Expected Output
  • Base GPT-2: generates generic, non-recipe text when prompted
  • Training loss: starts ~3.5, should drop to ~1.8 by epoch 3
  • Fine-tuned: generates ingredient lists and cooking steps in recipe format

PYTHON — Hyperparameter sensitivity experiment 
# ============================================================
# PRACTICE PROBLEM 3: Hyperparameter Sensitivity
# Uses DistilBERT on SST-2 (sentiment classification)
# ============================================================

!pip install transformers datasets evaluate scikit-learn -q

from transformers import (
    AutoTokenizer, AutoModelForSequenceClassification,
    TrainingArguments, Trainer
)
from datasets import load_dataset
import evaluate, numpy as np
import matplotlib.pyplot as plt

# CELL 1 — Load data (SST-2: binary sentiment)
dataset = load_dataset("glue", "sst2")
tokenizer = AutoTokenizer.from_pretrained("distilbert-base-uncased")

def tokenize(batch):
    return tokenizer(batch["sentence"], truncation=True, padding="max_length", max_length=128)

# Use small subset for speed (500 train, 200 val)
small_train = dataset["train"].shuffle(seed=42).select(range(500)).map(tokenize, batched=True)
small_val   = dataset["validation"].shuffle(seed=42).select(range(200)).map(tokenize, batched=True)
small_train.set_format("torch", columns=["input_ids", "attention_mask", "label"])
small_val.set_format("torch", columns=["input_ids", "attention_mask", "label"])

metric = evaluate.load("accuracy")
def compute_metrics(eval_pred):
    logits, labels = eval_pred
    preds = np.argmax(logits, axis=-1)
    return metric.compute(predictions=preds, references=labels)

# CELL 2 — Run experiment across 4 learning rates
learning_rates = [1e-3, 1e-4, 5e-5, 1e-6]
results = {}

for lr in learning_rates:
    print(f"\n▶ Training with LR = {lr}")
    model = AutoModelForSequenceClassification.from_pretrained(
        "distilbert-base-uncased", num_labels=2
    )
    args = TrainingArguments(
        output_dir              = f"./lr-{lr}",
        num_train_epochs        = 3,
        per_device_train_batch_size = 16,
        per_device_eval_batch_size  = 32,
        learning_rate           = lr,
        warmup_steps            = 50,
        evaluation_strategy     = "epoch",
        logging_steps           = 20,
        load_best_model_at_end  = True,
        metric_for_best_model   = "accuracy",
        report_to               = "none"
    )
    trainer = Trainer(
        model           = model,
        args            = args,
        train_dataset   = small_train,
        eval_dataset    = small_val,
        compute_metrics = compute_metrics,
    )
    trainer.train()
    eval_result = trainer.evaluate()
    results[lr] = {
        "accuracy"  : eval_result["eval_accuracy"],
        "loss"      : eval_result["eval_loss"],
        "log_history": trainer.state.log_history
    }
    print(f"  Final accuracy: {eval_result['eval_accuracy']:.4f}")

# CELL 3 — Plot results comparison
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 5))

# Bar chart: final accuracy per LR
lrs_str = [f"{lr}" for lr in learning_rates]
accs    = [results[lr]["accuracy"] for lr in learning_rates]
colors  = ['tomato', 'steelblue', 'seagreen', 'orange']
ax1.bar(lrs_str, accs, color=colors, edgecolor='white')
ax1.set_xlabel("Learning Rate"); ax1.set_ylabel("Validation Accuracy")
ax1.set_title("Final Accuracy vs Learning Rate")
ax1.set_ylim(0.5, 1.0)
for i, v in enumerate(accs):
    ax1.text(i, v + 0.005, f"{v:.3f}", ha='center', fontsize=9)

# Summary table
ax2.axis('off')
table_data = [[lr, f"{results[lr]['accuracy']:.4f}", f"{results[lr]['loss']:.4f}"] for lr in learning_rates]
tbl = ax2.table(
    cellText   = table_data,
    colLabels  = ["Learning Rate", "Val Accuracy", "Val Loss"],
    loc        = 'center', cellLoc='center'
)
tbl.scale(1.4, 2.0)
ax2.set_title("Results Summary")
plt.tight_layout()
plt.show()

best_lr = max(results, key=lambda x: results[x]["accuracy"])
print(f"\n🏆 Best learning rate: {best_lr} → accuracy: {results[best_lr]['accuracy']:.4f}")
Expected Output
  • LR=1e-3: unstable, low accuracy (~55-60%) — too aggressive
  • LR=5e-5: best performance (~85-90%) — standard fine-tuning range
  • LR=1e-6: barely changes from base model (~50-55%) — too slow
  • Bar chart clearly showing the "Goldilocks zone" of learning rates

# ============================================================
# PRACTICE PROBLEM 4: BLEU & ROUGE Evaluation
# No GPU needed — runs on CPU in ~5 minutes
# ============================================================

!pip install transformers datasets evaluate rouge-score sacrebleu -q

from transformers import pipeline
from datasets import load_dataset
import evaluate
import pandas as pd
import matplotlib.pyplot as plt

# CELL 1 — Understand BLEU manually first
def get_ngrams(text, n):
    words = text.lower().split()
    return [tuple(words[i:i+n]) for i in range(len(words) - n + 1)]

reference = "The cat sat on the mat"
generated = "The cat is on the mat"

for n in [1, 2, 3]:
    ref_ng = set(get_ngrams(reference, n))
    gen_ng = get_ngrams(generated, n)
    matches = sum(1 for ng in gen_ng if ng in ref_ng)
    precision = matches / len(gen_ng) if gen_ng else 0
    print(f"BLEU-{n} precision: {matches}/{len(gen_ng)} = {precision:.3f}")

# CELL 2 — Load a small dialogue summarization test set
dataset = load_dataset("knkarthick/dialogsum", split="test[:50]")
print(f"Test samples: {len(dataset)}")

# CELL 3 — Use Flan-T5 to generate summaries
summarizer = pipeline(
    "text2text-generation",
    model     = "google/flan-t5-base",
    tokenizer = "google/flan-t5-base",
    max_length = 128
)

print("Generating summaries (this may take 2-3 minutes)...")
predictions = []
references  = []

for i, item in enumerate(dataset):
    prompt = f"Summarize the following conversation:\n{item['dialogue']}"
    output = summarizer(prompt)[0]['generated_text']
    predictions.append(output)
    references.append(item['summary'])
    if i % 10 == 0: print(f"  {i+1}/50 done")

# CELL 4 — Compute BLEU
bleu = evaluate.load("sacrebleu")
bleu_score = bleu.compute(
    predictions = predictions,
    references  = [[r] for r in references]
)
print(f"\n📊 BLEU score: {bleu_score['score']:.2f} / 100")

# CELL 5 — Compute ROUGE
rouge = evaluate.load("rouge")
rouge_score = rouge.compute(
    predictions = predictions,
    references  = references
)
print(f"📊 ROUGE-1 : {rouge_score['rouge1']:.4f}")
print(f"📊 ROUGE-2 : {rouge_score['rouge2']:.4f}")
print(f"📊 ROUGE-L : {rouge_score['rougeL']:.4f}")

# CELL 6 — Show side-by-side examples
print("\n=== QUALITATIVE EXAMPLES ===")
for i in [0, 5, 10]:
    print(f"\n[Example {i+1}]")
    print(f"Dialogue : {dataset[i]['dialogue'][:120]}...")
    print(f"Reference: {references[i]}")
    print(f"Generated: {predictions[i]}")

# CELL 7 — Visualize scores
labels = ['ROUGE-1', 'ROUGE-2', 'ROUGE-L']
values = [rouge_score['rouge1'], rouge_score['rouge2'], rouge_score['rougeL']]
colors = ['#22d3a0', '#60a5fa', '#a78bfa']
plt.figure(figsize=(8, 4))
bars = plt.bar(labels, values, color=colors, edgecolor='white', width=0.5)
plt.ylim(0, 1)
plt.title("ROUGE Scores — Flan-T5 on DialogSum (50 test samples)")
plt.ylabel("Score")
for bar, val in zip(bars, values):
    plt.text(bar.get_x() + bar.get_width()/2, val + 0.01, f"{val:.3f}", ha='center')
plt.tight_layout()
plt.show()
Expected Output
  • BLEU score: ~15-25 (reasonable for a base model, not fine-tuned)
  • ROUGE-1: ~0.35–0.45, ROUGE-2: ~0.12–0.18, ROUGE-L: ~0.30–0.40
  • Qualitative examples showing where the model gets it right vs wrong

# ============================================================
# PRACTICE PROBLEM 5: Build a RAG Pipeline
# No GPU needed. Uses FAISS + Sentence-BERT + Flan-T5
# ============================================================

!pip install transformers sentence-transformers faiss-cpu -q

from sentence_transformers import SentenceTransformer
from transformers import pipeline
import faiss, numpy as np

# CELL 1 — Your "knowledge base" (replace with your own documents!)
documents = [
    "PayPal is a digital payments platform founded in 1998. It allows users to send and receive money online.",
    "Buy Now Pay Later (BNPL) services let customers purchase items immediately and pay in installments.",
    "BNPL products typically charge 0% interest if paid on time, but late fees can apply.",
    "PayPal Pay Later is PayPal's BNPL offering, available at millions of merchants.",
    "Fine-tuning an LLM means continuing training on a smaller, domain-specific dataset.",
    "Retrieval Augmented Generation (RAG) combines a retriever with a language model to produce factual answers.",
    "BLEU score measures machine translation quality by comparing n-gram overlap with reference translations.",
    "ROUGE score is used for summarization evaluation, measuring recall of n-grams in generated text.",
    "The learning rate in neural network training controls how much weights are updated per gradient step.",
    "Overfitting occurs when a model memorizes training data and fails to generalize to new examples.",
    "Transformer architecture uses self-attention to weigh the importance of different words in context.",
    "GPT-2 is a generative pre-trained transformer with 1.5 billion parameters trained on WebText dataset.",
    "Flan-T5 is an instruction-tuned version of T5, trained on a mixture of tasks for better zero-shot performance.",
    "Vector embeddings convert text into numerical arrays that capture semantic meaning.",
    "FAISS (Facebook AI Similarity Search) enables fast nearest-neighbor search over large sets of vectors.",
]

# CELL 2 — Embed all documents into a FAISS vector index
embedder = SentenceTransformer("all-MiniLM-L6-v2")  # Fast, small, good quality

print("Embedding documents...")
doc_embeddings = embedder.encode(documents, convert_to_numpy=True)
print(f"Embedding shape: {doc_embeddings.shape}")  # (15, 384)

# Build FAISS index
dim   = doc_embeddings.shape[1]
index = faiss.IndexFlatL2(dim)
index.add(doc_embeddings.astype('float32'))
print(f"FAISS index built with {index.ntotal} vectors")

# CELL 3 — Retriever function
def retrieve(query, top_k=3):
    query_vec = embedder.encode([query], convert_to_numpy=True).astype('float32')
    distances, indices = index.search(query_vec, top_k)
    retrieved = [documents[i] for i in indices[0]]
    return retrieved, distances[0]

# CELL 4 — Load Flan-T5 as the generator
generator = pipeline("text2text-generation", model="google/flan-t5-base", max_length=200)

# CELL 5 — The full RAG function
def rag_answer(query, top_k=3):
    # Step 1: Retrieve
    context_docs, scores = retrieve(query, top_k=top_k)
    context = "\n".join(f"- {doc}" for doc in context_docs)

    # Step 2: Augment prompt with context
    prompt = f"""Based on the following context, answer the question.

Context:
{context}

Question: {query}

Answer:"""

    # Step 3: Generate
    response = generator(prompt)[0]['generated_text']
    return response, context_docs

def vanilla_answer(query):
    # Without RAG — just the raw question
    return generator(query)[0]['generated_text']

# CELL 6 — Test with real questions
test_questions = [
    "What is BNPL and how does PayPal offer it?",
    "What is the difference between BLEU and ROUGE?",
    "Why does overfitting happen and how does it affect model performance?",
    "What is FAISS used for in a RAG system?",
]

print("="*60)
for q in test_questions:
    rag_ans, ctx = rag_answer(q)
    van_ans = vanilla_answer(q)
    print(f"\n❓ Question: {q}")
    print(f"🤖 Vanilla : {van_ans}")
    print(f"📚 RAG     : {rag_ans}")
    print(f"📄 Sources : {ctx[0][:80]}...")
    print("-"*60)

# CELL 7 — Interactive mode: ask your own questions!
print("\n🎯 Interactive Mode — Ask anything from the knowledge base!")
print("Type 'quit' to stop\n")
while True:
    q = input("Your question: ")
    if q.lower() == 'quit': break
    ans, ctx = rag_answer(q)
    print(f"Answer: {ans}")
    print(f"Based on: {ctx[0][:100]}...\n")
Expected Output
  • Embedding shape: (15, 384) — 15 documents, 384-dim vectors
  • Vanilla answer: generic / sometimes wrong without context
  • RAG answer: directly references facts from the knowledge base
  • Interactive mode: answers questions using retrieved context
